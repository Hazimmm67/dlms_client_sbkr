cmake_minimum_required(VERSION 3.16)
project(dlms_client_sbkr VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CZMQ REQUIRED IMPORTED_TARGET libczmq)
set(GURUX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/gurux/development" CACHE PATH "Gurux development source directory")
file(GLOB GURUX_SOURCES CONFIGURE_DEPENDS "${GURUX_ROOT}/src/*.c")
add_library(dlms_sbkr_gurux STATIC ${GURUX_SOURCES} vendor/transport/src/gurux_callbacks.c)
target_include_directories(dlms_sbkr_gurux PRIVATE vendor/transport/include)
target_include_directories(dlms_sbkr_gurux PUBLIC "${GURUX_ROOT}/include")
add_library(dlms_sbkr_json STATIC vendor/cjson/cJSON.c)
set_target_properties(dlms_sbkr_json PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dlms_sbkr_json PUBLIC vendor/cjson)
add_library(dlms_client_sbkr STATIC src/config.c src/dlms_client_sbkr.c src/transport.c src/gateway.c
    vendor/transport/src/communication.c vendor/transport/src/connection.c)
add_library(dlms_client_sbkr::client ALIAS dlms_client_sbkr)
target_include_directories(dlms_client_sbkr PUBLIC include vendor/sbkr PRIVATE src vendor/transport/include)
target_link_libraries(dlms_client_sbkr PUBLIC PkgConfig::CZMQ PRIVATE dlms_sbkr_gurux dlms_sbkr_json m ${CMAKE_DL_LIBS})
target_compile_definitions(dlms_client_sbkr PRIVATE _DEFAULT_SOURCE _POSIX_C_SOURCE=200809L)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set_source_files_properties(src/config.c src/dlms_client_sbkr.c src/transport.c src/gateway.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic")
endif()
set(DLMS_SBKR_TOOLS_DEFAULT OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(DLMS_SBKR_TOOLS_DEFAULT ON)
endif()
option(DLMS_SBKR_BUILD_TOOLS "Build CLI and development meter" ${DLMS_SBKR_TOOLS_DEFAULT})
if(DLMS_SBKR_BUILD_TOOLS)
add_executable(dlms_client_sbkr_cli src/main.c)
set_target_properties(dlms_client_sbkr_cli PROPERTIES OUTPUT_NAME dlms_client_sbkr)
target_include_directories(dlms_client_sbkr_cli PRIVATE src vendor/transport/include)
target_link_libraries(dlms_client_sbkr_cli PRIVATE dlms_client_sbkr dlms_sbkr_json dlms_sbkr_gurux)
add_executable(dlms_meter_simulator src/meter_simulator.c)
target_include_directories(dlms_meter_simulator PRIVATE src vendor/transport/include)
target_link_libraries(dlms_meter_simulator PRIVATE dlms_client_sbkr dlms_sbkr_gurux)
endif()
option(DLMS_SBKR_BUILD_DEMO "Build the temporary local SBKR demo library and gateway" ${DLMS_SBKR_TOOLS_DEFAULT})
if(DLMS_SBKR_BUILD_DEMO)
    add_library(sbkr_demo SHARED demo/library.c demo/protocol.c)
    target_include_directories(sbkr_demo PRIVATE demo vendor/sbkr)
    target_link_libraries(sbkr_demo PRIVATE dlms_sbkr_json PkgConfig::CZMQ m)
    target_compile_definitions(sbkr_demo PRIVATE _POSIX_C_SOURCE=200809L)
    add_executable(sbkr_demo_gateway demo/gateway.c demo/protocol.c)
    target_include_directories(sbkr_demo_gateway PRIVATE demo vendor/sbkr)
    target_link_libraries(sbkr_demo_gateway PRIVATE dlms_sbkr_json PkgConfig::CZMQ m)
    target_compile_definitions(sbkr_demo_gateway PRIVATE _POSIX_C_SOURCE=200809L)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(sbkr_demo PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(sbkr_demo_gateway PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
option(DLMS_SBKR_BUILD_TESTS "Build offline integration tests" OFF)
if(DLMS_SBKR_BUILD_TESTS)
    if(NOT DLMS_SBKR_BUILD_TOOLS)
        message(FATAL_ERROR "Tests require DLMS_SBKR_BUILD_TOOLS=ON")
    endif()
    enable_testing()
    add_library(fake_sbkr SHARED tests/fake_sbkr.c)
    target_include_directories(fake_sbkr PRIVATE vendor/sbkr)
    target_link_libraries(fake_sbkr PRIVATE PkgConfig::CZMQ)
    add_executable(test_gateway tests/test_gateway.c)
    target_link_libraries(test_gateway PRIVATE dlms_client_sbkr)
    add_test(NAME gateway_adapter COMMAND test_gateway $<TARGET_FILE:fake_sbkr>)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    if(DLMS_SBKR_BUILD_DEMO)
        add_executable(test_demo_api tests/test_demo_api.c)
        target_include_directories(test_demo_api PRIVATE vendor/sbkr)
        target_link_libraries(test_demo_api PRIVATE sbkr_demo PkgConfig::CZMQ m)
        add_test(NAME demo_sbkr_integration COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demo_integration.py"
            $<TARGET_FILE:dlms_client_sbkr_cli> $<TARGET_FILE:dlms_meter_simulator>
            $<TARGET_FILE:sbkr_demo_gateway> $<TARGET_FILE:sbkr_demo> $<TARGET_FILE:test_demo_api>
            "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.tcp-demo.json")
        set_tests_properties(demo_sbkr_integration PROPERTIES TIMEOUT 45)
        add_test(NAME demo_runner_recovery COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demo_runner.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/run_sbkr_demo.py" "${CMAKE_CURRENT_BINARY_DIR}")
        set_tests_properties(demo_runner_recovery PROPERTIES TIMEOUT 45)
    endif()
    add_test(NAME tcp_roundtrip COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tcp.py" $<TARGET_FILE:dlms_client_sbkr_cli> $<TARGET_FILE:dlms_meter_simulator> "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.tcp-demo.json")
    set_tests_properties(tcp_roundtrip PROPERTIES TIMEOUT 35)
    add_test(NAME config_errors COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config.py" $<TARGET_FILE:dlms_client_sbkr_cli> "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.json")
    add_executable(test_client tests/test_client.c)
    target_link_libraries(test_client PRIVATE dlms_client_sbkr)
    add_test(NAME client COMMAND test_client "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.json")
    add_test(NAME config COMMAND dlms_client_sbkr_cli "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.json" --check)
    add_test(NAME simulated_poll COMMAND dlms_client_sbkr_cli "${CMAKE_CURRENT_SOURCE_DIR}/config/dtm.json" --simulate --once)
endif()
